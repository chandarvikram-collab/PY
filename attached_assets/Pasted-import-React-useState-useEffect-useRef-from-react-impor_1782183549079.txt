/* ── Exercise Picker ── */
function ExercisePicker({ onAdd, onClose, currentIds }) {
  const [q,setQ]=useState('');const [cat,setCat]=useState('All');const [eq,setEq]=useState('All');
  const filtered=EXERCISES.filter(ex=>{
    if(cat!=='All'&&ex.category!==cat)return false;
    if(eq!=='All'&&ex.equipment!==eq)return false;
    if(q&&!ex.name.toLowerCase().includes(q.toLowerCase()))return false;
    return true;
  });
  return(
    <div className="overlay"><div className="sheet slide">
      <div style={{display:'flex',justifyContent:'space-between',alignItems:'center',marginBottom:14}}>
        <span className="h2">Add Exercise</span>
        <button onClick={onClose} style={{background:'none',border:'none',color:'#6B6B68',cursor:'pointer'}}><X size={20}/></button>
      </div>
      <input className="inp" value={q} onChange={e=>setQ(e.target.value)} placeholder="Search..." style={{marginBottom:8}}/>
      <div className="cat-scr">{CATEGORIES.map(c=><button key={c} className={`chip ${cat===c?'on':''}`} onClick={()=>setCat(c)}>{c}</button>)}</div>
      <div className="cat-scr">{EQUIPMENTS.map(e=><button key={e} className={`chip ${eq===e?'on':''}`} style={{fontSize:11,padding:'5px 10px'}} onClick={()=>setEq(e)}>{e}</button>)}</div>
      <div style={{overflowY:'auto',maxHeight:300}}>
        {filtered.length===0&&<div style={{color:'#6B6B68',textAlign:'center',padding:'20px 0',fontSize:13}}>No matches.</div>}
        {filtered.map(ex=>{const added=currentIds.includes(ex.id);return(
          <div key={ex.id} style={{display:'flex',alignItems:'center',gap:10,padding:'9px 12px',background:'#1E1E1E',borderRadius:10,border:'1px solid rgba(255,255,255,0.04)',marginBottom:6}}>
            <div style={{flex:1}}>
              <div style={{display:'flex',gap:5,marginBottom:3}}><span className="pill" style={{background:'#2A1414',color:A}}>{ex.category}</span><span className="pill">{ex.equipment}</span></div>
              <div style={{color:'#F5F4F1',fontWeight:700,fontSize:13}}>{ex.name}</div>
            </div>
            <button onClick={()=>{if(!added){onAdd(ex);onClose();}}} style={{width:30,height:30,borderRadius:'50%',border:added?'none':'1px solid #3A3A38',background:added?'#2A1414':'transparent',color:'#F5F4F1',display:'flex',alignItems:'center',justifyContent:'center',cursor:added?'default':'pointer',flexShrink:0}}>
              {added?<Check size={14} color={A}/>:<Plus size={14}/>}
            </button>
          </div>
        );})}
      </div>
    </div></div>
  );
}

function ValueNumpad({ label, initialVal, onConfirm, onClose }) {
  const [val,setVal]=useState(initialVal||'');const isW=label==='lbs';
  const press=k=>{
    if(k==='DEL'){setVal(v=>v.length>1?v.slice(0,-1):'');return;}
    if(k==='.'){setVal(v=>v.includes('.')?v:(v===''?'0':v)+'.');return;}
    if(k==='+2.5'||k==='-2.5'){setVal(v=>String(Math.max(0,(parseFloat(v)||0)+(k[0]==='+'?2.5:-2.5))));return;}

  };
  return(
    <div className="overlay" style={{zIndex:200}}><div className="sheet slide">
      <div style={{display:'flex',justifyContent:'space-between',marginBottom:10}}>
        <span style={{color:'#8C8C89',fontSize:12,fontWeight:700,textTransform:'uppercase',letterSpacing:1}}>{label}</span>

      </div>
      <div style={{textAlign:'center',fontFamily:'Anton',fontSize:52,color:'#F5F4F1',marginBottom:14}}>

/* ── Gym Session ── */
function GymSession({ routine, onFinish }) {
  const buildSets=ex=>{const prev=PREV_PERF[ex.name]||null;return Array.from({length:ex.sets||3},(_,i)=>({setNum:i+1,prevWeight:prev?prev.weight:null,prevReps:prev?prev.reps:null,weight:String(ex.weight||''),reps:String(ex.reps||''),done:false,restTotal:ex.rest||120}));};
  const [exercises,setExercises]=useState(()=>(routine.exercises||[]).map(ex=>({...ex,uid:ex.id+'-'+Math.random(),sets:buildSets(ex)})));
  const [elapsed,setElapsed]=useState(0);const [finished,setFinished]=useState(false);
  const [showPicker,setShowPicker]=useState(false);const [numpad,setNumpad]=useState(null);const [rest,setRest]=useState(null);
  useEffect(()=>{const t=setInterval(()=>setElapsed(s=>s+1),1000);return()=>clearInterval(t);},[]);
  useEffect(()=>{if(!rest||rest.paused||rest.left<=0)return;const t=setTimeout(()=>setRest(r=>r?{...r,left:r.left-1}:null),1000);return()=>clearTimeout(t);},[rest]);
  const updSet=(uid,si,patch)=>setExercises(prev=>prev.map(ex=>ex.uid!==uid?ex:{...ex,sets:ex.sets.map((s,i)=>i!==si?s:{...s,...patch})}));
  const checkSet=(uid,si)=>{const ex=exercises.find(e=>e.uid===uid);const s=ex.sets[si];if(s.done){updSet(uid,si,{done:false});if(rest&&rest.uid===uid&&rest.si===si)setRest(null);return;}updSet(uid,si,{done:true});setRest({uid,si,total:s.restTotal,left:s.restTotal,paused:false});};
  const addSet=uid=>setExercises(prev=>prev.map(ex=>{if(ex.uid!==uid)return ex;const last=ex.sets[ex.sets.length-1];return{...ex,sets:[...ex.sets,{setNum:ex.sets.length+1,prevWeight:null,prevReps:null,weight:last?last.weight:'',reps:last?last.reps:'',done:false,restTotal:120}]};}));
  const rmEx=uid=>setExercises(prev=>prev.filter(ex=>ex.uid!==uid));
  const addExercise=ex=>setExercises(prev=>[...prev,{...ex,uid:ex.id+'-'+Math.random(),sets:buildSets({...ex,sets:3,reps:10,weight:0,rest:120})}]);
  const completedSets=exercises.reduce((s,ex)=>s+ex.sets.filter(s=>s.done).length,0);
  const totalSets=exercises.reduce((s,ex)=>s+ex.sets.length,0);
  const totalVolume=exercises.reduce((sum,ex)=>sum+ex.sets.filter(s=>s.done).reduce((sv,s)=>sv+(parseFloat(s.weight)||0)*(parseInt(s.reps)||0),0),0);

  if(finished){
    const exerciseLog=exercises.map(ex=>({name:ex.name,category:ex.category,sets:ex.sets.filter(s=>s.done).map(s=>({weight:parseFloat(s.weight)||0,reps:parseInt(s.reps)||0}))}));
    const summary={id:'h-'+Date.now(),type:'gym',name:routine.name,date:todayKey(),duration:elapsed,exercises:exercises.length,volume:Math.round(totalVolume),exerciseLog};
    return(
      <div className="fade" style={{textAlign:'center',paddingTop:32,paddingBottom:32}}>
        <div style={{width:64,height:64,background:'#2A1414',borderRadius:'50%',display:'flex',alignItems:'center',justifyContent:'center',margin:'0 auto 16px'}}><Trophy size={28} color={A}/></div>
        <h1 className="h1" style={{marginBottom:6}}>Done!</h1>
        <p style={{color:'#8C8C89',fontSize:13,marginBottom:24}}>{routine.name} · {fmtTime(elapsed)}</p>
        <div style={{display:'grid',gridTemplateColumns:'1fr 1fr 1fr',gap:10,marginBottom:20}}>
          {[{l:'Duration',v:fmtTime(elapsed)},{l:'Volume',v:`${Math.round(totalVolume).toLocaleString()} lbs`},{l:'Sets',v:`${completedSets}/${totalSets}`}].map(s=>(
            <div key={s.l} className="card" style={{padding:12,textAlign:'center'}}><div style={{color:'#F5F4F1',fontFamily:'Anton',fontSize:14}}>{s.v}</div><div style={{color:'#6B6B68',fontSize:10,marginTop:2}}>{s.l}</div></div>
          ))}
        </div>
        <button className="btn-p" onClick={()=>onFinish(summary)}>Back to Home</button>
      </div>
    );
  }

  return(
    <div>
      {showPicker&&<ExercisePicker onAdd={ex=>{addExercise(ex);setShowPicker(false);}} onClose={()=>setShowPicker(false)} currentIds={exercises.map(e=>e.id)}/>}
      {numpad&&<ValueNumpad label={numpad.field==='weight'?'lbs':'reps'} initialVal={exercises.find(e=>e.uid===numpad.uid)?.sets[numpad.si]?.[numpad.field]||''} onConfirm={val=>{updSet(numpad.uid,numpad.si,{[numpad.field]:val});setNumpad(null);}} onClose={()=>setNumpad(null)}/>}
      <div style={{display:'flex',justifyContent:'space-between',alignItems:'center',marginBottom:10}}>
        <div><p className="eyebrow" style={{marginBottom:2}}>Active Workout</p><h2 className="h2" style={{fontSize:17}}>{routine.name}</h2></div>
        <div style={{textAlign:'right'}}><div style={{fontFamily:'Anton',fontSize:22,color:A}}>{fmtTime(elapsed)}</div><div style={{color:'#6B6B68',fontSize:10}}>{completedSets}/{totalSets} sets</div></div>
      </div>
      <div style={{display:'flex',gap:8,marginBottom:14}}>
        {[{l:'Volume',v:`${Math.round(totalVolume).toLocaleString()} lbs`},{l:'Exercises',v:exercises.length}].map(s=>(
          <div key={s.l} style={{flex:1,background:'#1A1A1A',borderRadius:10,padding:'8px 10px',textAlign:'center'}}><div style={{color:'#F5F4F1',fontFamily:'Anton',fontSize:14}}>{s.v}</div><div style={{color:'#6B6B68',fontSize:9}}>{s.l}</div></div>
        ))}
        <button className="btn-p" style={{flex:1,padding:'8px',fontSize:11,borderRadius:10}} onClick={()=>setFinished(true)}>Finish</button>
      </div>
      {exercises.map(ex=>(
        <div key={ex.uid} className="card" style={{marginBottom:12}}>
          <div style={{display:'flex',justifyContent:'space-between',alignItems:'flex-start',marginBottom:10}}>
            <div><div style={{color:A,fontWeight:800,fontSize:14,marginBottom:3}}>{ex.name}</div><div style={{display:'flex',gap:5}}><span className="pill">{ex.category}</span><span className="pill">{ex.equipment}</span></div></div>
            <button onClick={()=>rmEx(ex.uid)} style={{background:'none',border:'none',color:'#6B6B68',cursor:'pointer'}}><X size={16}/></button>
          </div>
          <div className="set-row" style={{marginBottom:6}}>
            <span style={{color:'#6B6B68',fontSize:11,fontWeight:700,textAlign:'center'}}>SET</span>
            <span style={{color:'#6B6B68',fontSize:11,fontWeight:700}}>PREVIOUS</span>
            <span style={{color:'#6B6B68',fontSize:11,fontWeight:700,textAlign:'center'}}>LBS</span>
            <span style={{color:'#6B6B68',fontSize:11,fontWeight:700,textAlign:'center'}}>REPS</span>
            <Check size={12} color="#6B6B68"/>
          </div>
          {ex.sets.map((s,si)=>(
            <div key={si} style={{marginBottom:4}}>
              <div className="set-row" style={{opacity:s.done?0.6:1}}>
                <div className="set-badge" style={{background:s.done?'#2A1414':'#2A2A28',color:s.done?A:'#ADADA9'}}>{s.setNum}</div>
                <span style={{color:'#6B6B68',fontSize:11}}>{s.prevWeight!=null?`${s.prevWeight}×${s.prevReps}`:'—'}</span>
                <div className={`set-cell${s.done?' done':''}`} onClick={()=>!s.done&&setNumpad({uid:ex.uid,si,field:'weight'})}><span style={{color:s.weight?'#F5F4F1':'#4A4A48',fontWeight:700,fontSize:14}}>{s.weight||'—'}</span></div>
                <div className={`set-cell${s.done?' done':''}`} onClick={()=>!s.done&&setNumpad({uid:ex.uid,si,field:'reps'})}><span style={{color:s.reps?'#F5F4F1':'#4A4A48',fontWeight:700,fontSize:14}}>{s.reps||'—'}</span></div>
                <button className="chk" onClick={()=>checkSet(ex.uid,si)} style={{background:s.done?A:'#2A2A28'}}><Check size={13} color={s.done?'#F5F4F1':'#6B6B68'}/></button>
              </div>
              {rest&&rest.uid===ex.uid&&rest.si===si&&rest.left>0?(
                <div>
                  <div className="rest-bar"><div className="rest-fill" style={{width:`${((rest.total-rest.left)/rest.total)*100}%`}}/><span className="rest-lbl">REST {fmtTime(rest.left)}</span></div>
                  <div style={{display:'flex',gap:6,marginBottom:4}}>
                    <button onClick={()=>setRest(r=>r?{...r,paused:!r.paused}:null)} style={{flex:1,background:'#1E1E1E',border:'1px solid #2A2A28',borderRadius:8,padding:'7px',color:'#ADADA9',fontSize:12,fontWeight:600,cursor:'pointer'}}>{rest.paused?'Resume':'Pause'}</button>
                    <button onClick={()=>setRest(null)} style={{flex:1,background:A,border:'none',borderRadius:8,padding:'7px',color:'#F5F4F1',fontSize:12,fontWeight:700,cursor:'pointer'}}>Skip</button>
                  </div>
                </div>
              ):(
                <div className="rest-static"><div className="rest-line"/><span style={{color:'#3A3A38',fontSize:10,padding:'0 6px'}}>{fmtTime(s.restTotal)}</span><div className="rest-line"/></div>
              )}
            </div>
          ))}
          <button onClick={()=>addSet(ex.uid)} style={{width:'100%',marginTop:4,background:'#141414',border:'1px dashed #3A3A38',borderRadius:8,padding:'9px',color:'#ADADA9',fontSize:12,fontWeight:600,cursor:'pointer'}}>+ Add Set</button>
        </div>
      ))}
      <button className="btn-g" onClick={()=>setShowPicker(true)} style={{marginBottom:10}}><Plus size={14}/> Add Exercise</button>
      <button className="btn-p" onClick={()=>setFinished(true)} style={{marginBottom:24}}><CheckCircle size={14}/> Finish Workout</button>
    </div>
  );
}

/* ── Mini SVG Route Map ── */
function MiniMap({ progress=0 }) {
  const pts=[[20,110],[70,75],[130,95],[190,55],[250,75],[330,35]];
  const pathD=pts=>{let d=`M${pts[0][0]},${pts[0][1]}`;for(let i=1;i<pts.length;i++){const cp=(pts[i-1][0]+pts[i][0])/2;d+=` Q${cp},${pts[i-1][1]} ${pts[i][0]},${pts[i][1]}`;}return d;};
  const n=Math.max(2,Math.floor(progress*(pts.length-1))+1);const live=pts.slice(0,n);const last=live[live.length-1];
  return(
    <svg viewBox="0 0 360 130" width="100%" height="100%" style={{display:'block'}}>
      <rect width="360" height="130" fill="#1A1A1A"/>
      {[1,2,3].map(i=><line key={i} x1={0} y1={i*35} x2={360} y2={i*35} stroke="rgba(255,255,255,0.03)" strokeWidth="1"/>)}
      <path d={pathD(pts)} stroke="rgba(232,21,27,0.1)" strokeWidth="2.5" fill="none" strokeDasharray="5 4"/>
      {progress>0&&<path d={pathD(live)} stroke={A} strokeWidth="2.5" fill="none" strokeLinecap="round" style={{filter:'drop-shadow(0 0 4px rgba(232,21,27,0.5))'}}/>}
      <circle cx={pts[0][0]} cy={pts[0][1]} r="5" fill="#22C55E"/>
      {progress>0&&<><circle cx={last[0]} cy={last[1]} r="14" fill="rgba(232,21,27,0.08)"/><circle cx={last[0]} cy={last[1]} r="5.5" fill={A} style={{filter:'drop-shadow(0 0 5px rgba(232,21,27,0.8))'}}/></>}
    </svg>
  );
}

function ElevationProfile({ elevGain }) {
  // Generate a pseudo-elevation profile from elevGain
  const points = [0,0.15,0.32,0.55,0.7,0.62,0.78,0.88,1.0,0.92,0.85].map((t,i)=>({
    x: i/(10)*320+20,
    y: 70 - t * Math.min(elevGain, 60)
  }));
  const pathD = points.map((p,i)=>i===0?`M${p.x},${p.y}`:`L${p.x},${p.y}`).join(' ');
  const fillD = pathD + ` L${points[points.length-1].x},80 L${points[0].x},80 Z`;
  return (
    <svg viewBox="0 0 360 90" width="100%" height="70" style={{display:'block'}}>
      <rect width="360" height="90" fill="#111"/>
      <path d={fillD} fill={`rgba(232,21,27,0.15)`}/>
      <path d={pathD} stroke={A} strokeWidth="2" fill="none" strokeLinecap="round" strokeLinejoin="round"/>
      <line x1="20" y1="80" x2="340" y2="80" stroke="#2A2A28" strokeWidth="1"/>
    </svg>
  );
}

export default function IronPaceApp() {
  const [tab,setTab]=useState('home');
  const [routines,setRoutines]=useState(SEED_ROUTINES);
  const [gymHistory,setGymHistory]=useState(SEED_GYM_HISTORY);
  const [runHistory,setRunHistory]=useState(SEED_RUN_HISTORY);
  const [volumeData,setVolumeData]=useState(SEED_VOLUME);
  const [runTrend,setRunTrend]=useState(SEED_RUN_TREND);
  const [completedDates,setCompletedDates]=useState(new Set());
  const [session,setSession]=useState(null);
  const [time,setTime]=useState(new Date());

  useEffect(()=>{const saved=loadState();if(!saved)return;if(saved.routines)setRoutines(saved.routines);if(saved.gymHistory)setGymHistory(saved.gymHistory);if(saved.runHistory)setRunHistory(saved.runHistory);if(saved.volumeData)setVolumeData(saved.volumeData);if(saved.runTrend)setRunTrend(saved.runTrend);if(saved.completedDates)setCompletedDates(new Set(saved.completedDates));},[]);
  useEffect(()=>{saveState({routines,gymHistory,runHistory,volumeData,runTrend,completedDates:[...completedDates]});},[routines,gymHistory,runHistory,volumeData,runTrend,completedDates]);
  useEffect(()=>{const t=setInterval(()=>setTime(new Date()),30000);return()=>clearInterval(t);},[]);

  const markToday=()=>setCompletedDates(prev=>new Set([...prev,todayKey()]));
  const finishGym=summary=>{setGymHistory(prev=>[summary,...prev]);setVolumeData(prev=>{if(!prev.length)return prev;const n=[...prev];n[n.length-1]={...n[n.length-1],volume:n[n.length-1].volume+summary.volume};return n;});markToday();setSession(null);setTab('home');};
  const finishRun=summary=>{setRunHistory(prev=>[summary,...prev]);setRunTrend(prev=>{if(!prev.length)return prev;const n=[...prev];n[n.length-1]={...n[n.length-1],km:+(n[n.length-1].km+summary.distKm).toFixed(1)};return n;});markToday();setSession(null);setTab('home');};

  return(
    <div className="wrap">
      <style>{css}</style>
      <div className="phone">
        <div className="statusbar">
          <span>{time.toLocaleTimeString('en-US',{hour:'numeric',minute:'2-digit',hour12:false})}</span>
          <span style={{fontFamily:'Anton',fontSize:10,letterSpacing:3,color:A}}>IRONPACE</span>
          <span style={{display:'flex',gap:4,alignItems:'center'}}><span style={{width:14,height:7,border:'1.5px solid #F5F4F1',borderRadius:2,position:'relative',display:'inline-block'}}><span style={{position:'absolute',inset:1,right:2,background:'#F5F4F1',borderRadius:1}}/></span></span>
        </div>
        <div className="screen">
          {session?.kind==='run-launcher'&&<RunLauncher onBack={()=>setSession(null)} onStart={actType=>setSession({kind:'run',actType})}/>}
          {session?.kind==='gym'&&<GymSession routine={session.routine} onFinish={finishGym}/>}
          {session?.kind==='run'&&<RunSession actType={session.actType} onFinish={finishRun}/>}
          {!session&&tab==='home'&&<HomeScreen routines={routines} gymHistory={gymHistory} runHistory={runHistory} completedDates={completedDates} onStartGym={r=>setSession({kind:'gym',routine:r})} onStartRun={()=>setSession({kind:'run-launcher'})} onCreateGym={()=>setSession({kind:'gym',routine:{id:'q-'+Date.now(),name:'Quick Workout',color:A,exercises:[]}})}/>}
          {!session&&tab==='progress'&&<ProgressScreen gymHistory={gymHistory} runHistory={runHistory} volumeData={volumeData} runTrend={runTrend} strengthData={SEED_STRENGTH}/>}
        </div>
        {!session&&(
          <div className="nav">
            {[{id:'home',label:'Home',icon:Home},{id:'progress',label:'Progress',icon:TrendingUp}].map(({id,label,icon:Icon})=>(
              <button key={id} className={`navi ${tab===id?'on':''}`} onClick={()=>setTab(id)}><Icon size={20}/>{label}</button>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
